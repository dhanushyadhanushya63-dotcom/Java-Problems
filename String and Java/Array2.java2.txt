import java.util.Scanner;
public class Main{
    public static void main(String[] args){
        Scanner sc=new Scanner(System.in);
        int num=sc.nextInt();
        int[] arr=new int[num];
        for(int i=0;i<num;i++){
            arr[i]=sc.nextInt();
        }
        for(int i=0;i<num;i++){
            int sum=0;
            for(int j=0;j<num;j++){
                if(arr[j]==arr[i]){
                    continue;
                }
                else{
                  sum+=arr[j];
                }
            }
            System.out.print(sum+" ");
        }
    }
}
