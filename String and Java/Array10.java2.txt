import java.util.Scanner;
public class Main{
    public static void main(String[] args){
        Scanner sc=new Scanner(System.in);
    int n=sc.nextInt();
    int arr[]=new int[n];
    for(int i=0; i<n; i++){
        arr[i] = sc.nextInt();
    } 
    for(int j=0;j<n-1;j++){
        if(arr[j]==arr[j+1]){
            arr[j]=arr[j]*2;
            arr[j+1]=0;
        }
    }
    int k=0;
    for(int l=0;l<n;l++){
        if(arr[l]!=0){
            arr[k]=arr[l];
            k++;
        }
    }

    for(int j=k;j<n;j++){
        arr[j]=0;
    }

    for(int m=0; m<n; m++){
        System.out.print(arr[m] +" ");
    }
    }
   
}
