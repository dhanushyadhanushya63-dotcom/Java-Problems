import java.util.Scanner;
public class Main{ 
    public static void main(String[]args){
        Scanner sc=new Scanner(System.in);
        int n=sc.nextInt();
        long arr[]=new long[n];
        for(int i=0;i<n;i++){
            arr[i]=sc.nextLong();
        }
        long min=arr[0];
        long secMin=Long.MAX_VALUE;
        for(int i=0;i<n;i++){
            if(arr[i]<min){
                secMin=min;
                min=arr[i];
            }
            else if(arr[i]<secMin && arr[i]!=min){
                secMin=arr[i];
            }
        }
        System.out.print(secMin);
    }
}
