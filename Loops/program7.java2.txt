import java.util.Scanner;
public class Main{
    public static void main(String[] args){
        Scanner sc=new Scanner(System.in);
        long num=sc.nextLong();
        long fact=1;
        while(num!=0){
            long digit=num%10;
            fact=fact*digit;
            num=num/10;
        }
        System.out.println(fact);
    }
}
