import java.util.Scanner;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[][] arr = new int[n][2];
        for (int i = 0; i < n; i++) {
            arr[i][0] = sc.nextInt();
            arr[i][1] = sc.nextInt();
        }
        for (int i = 0; i < n - 1; i++) {
            for (int j = 0; j < n - i - 1; j++) {
                if (arr[j][0] > arr[j + 1][0]) {
                    int t1 = arr[j][0];
                    int t2 = arr[j][1];
                    arr[j][0] = arr[j + 1][0];
                    arr[j][1] = arr[j + 1][1];
                    arr[j + 1][0] = t1;
                    arr[j + 1][1] = t2;
                }
            }
        }
        int start = arr[0][0];
        int end = arr[0][1];

        for (int i = 1; i < n; i++) {
            if (arr[i][0] <= end) {
                if (arr[i][1] > end) {
                    end = arr[i][1];
                }
            } else {
                System.out.println(start + " " + end);
                start = arr[i][0];
                end = arr[i][1];
            }
        }
        System.out.println(start + " " + end);
    }
}
